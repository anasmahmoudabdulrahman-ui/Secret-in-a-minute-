function cleanSvg(raw) {
    let s = raw.replace(/```(?:svg|xml)?/g, '').trim();
    const i = s.indexOf('<svg');
    const j = s.lastIndexOf('</svg>');
    if (i < 0 || j < 0)
        return '';
    s = s.slice(i, j + 6);
    s = s.replace(/<script[\s\S]*?<\/script>/gi, '').replace(/\son\w+="[^"]*"/gi, '').replace(/<foreignObject[\s\S]*?<\/foreignObject>/gi, '');
    if (!/xmlns=/.test(s))
        s = s.replace('<svg', '<svg xmlns="http://www.w3.org/2000/svg"');
    if (!/viewBox=/.test(s))
        s = s.replace('<svg', '<svg viewBox="0 0 640 640"');
    s = s.replace(/<svg([^>]*?)\swidth="[^"]*"/, '<svg$1').replace(/<svg([^>]*?)\sheight="[^"]*"/, '<svg$1 width="640" height="640"');
    return s;
}

function b64ToF32(b64) {
    const bin = atob(b64);
    const n = Math.floor(bin.length / 2);
    const out = new Float32Array(n);
    for (let i = 0; i < n; i++) {
        let v = bin.charCodeAt(2 * i) | (bin.charCodeAt(2 * i + 1) << 8);
        if (v >= 0x8000)
            v -= 0x10000;
        out[i] = v / 32768;
    }
    return out;
}

async function callOnce(model, key, body) {
    let res;
    try {
        res = await fetch(`${API}${model}:generateContent`, { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-goog-api-key': key }, body: JSON.stringify(body) });
    }
    catch {

    }
    if (res.ok)
        return { ok: true, status: 200, data: await res.json() };
    return { ok: false, status: res.status, text: await res.text() };
}

async function discover(key, kind) {

// Retries each model with backoff, then tries the fallback models.
async function post(models, key, body, kind) {
    let lastStatus = 0, lastText = '';
    let list = preferred[kind] ? [preferred[kind], ...models.filter((m) => m !== preferred[kind])] : [...models];
    let extended = false;
    for (let mi = 0; mi < list.length; mi++) {
        const model = list[mi];
        const waits = [0, 4000, 9000];
        for (let t = 0; t < waits.length; t++) {
            if (waits[t]) {
                onStatus(`جوجل مشغولة، بحاول تاني (${t + 1} من ${waits.length}) بموديل ${model}…`);
                await sleep(waits[t]);
            }
            const r = await callOnce(model, key, body);
            if (r.ok) {
                preferred[kind] = model;
                return r.data;
            }
            lastStatus = r.status;
            lastText = r.text ?? '';
            const retry = r.status === 0 || r.status === 429 || r.status === 500 || r.status === 503 || r.status === 504;
            if (!retry)
                break;
        }
        if (lastStatus === 400 || lastStatus === 401 || lastStatus === 403)
            throw new Error(niceError(lastStatus, lastText));
        if (mi === list.length - 1 && !extended) {
            extended = true;
            discovered = discovered ?? discover(key, kind);
            const more = (await discovered).filter((n) => !list.includes(n)).slice(0, 3);
            list = [...list, ...more];
        }
        if (mi < list.length - 1)
            onStatus('بجرب موديل بديل…');
    }
    throw new Error(lastStatus === 0 ? 'مفيش اتصال بالنت أو جوجل مش بترد. جرب تاني.' : niceError(lastStatus, lastText));
}

function wrap(ctx, words, maxW) {
    const lines = [];
    let cur = [];
    for (const w of words) {
        const test = [...cur, w].join(' ');
        if (cur.length && ctx.measureText(test).width > maxW) {
            lines.push(cur);
            cur = [w];
        }
        else
            cur.push(w);
    }
    if (cur.length)
        lines.push(cur);
    return lines;
}

async function renderVideo(canvas, scenes, onProgress) {
    const ctx = canvas.getContext('2d');
    const imgs = await Promise.all(scenes.map((s) => loadSvg(s.draw)));
    const AC = window.AudioContext || window.webkitAudioContext;
    const ac = new AC();
    await Promise.race([ac.resume(), new Promise((r) => setTimeout(r, 1500))]);
    if (ac.state !== 'running')
        throw new Error('الصوت مش شغال في المتصفح. اضغط الزرار تاني.');
    const dest = ac.createMediaStreamDestination();
    const durs = scenes.map((s) => (s.pcm ? s.pcm.length / SR : 3) + 0.45);
    const starts = [];
    let acc = 0.3;
    for (const d of durs) {
        starts.push(acc);
        acc += d;
    }
    const total = acc + 0.3;
    const t0 = ac.currentTime + 0.2;
    scenes.forEach((s, i) => {
        if (!s.pcm || !s.pcm.length)
            return;
        const buf = ac.createBuffer(1, s.pcm.length, SR);
        buf.copyToChannel(s.pcm, 0);
        const src = ac.createBufferSource();
        src.buffer = buf;
        src.connect(dest);
        src.start(t0 + starts[i]);
    });
    const cands = ['video/mp4;codecs=avc1.42E01E,mp4a.40.2', 'video/mp4', 'video/webm;codecs=vp9,opus', 'video/webm;codecs=vp8,opus', 'video/webm'];
    const mime = cands.find((m) => window.MediaRecorder?.isTypeSupported?.(m)) ?? '';
    const stream = new MediaStream([...canvas.captureStream(30).getVideoTracks(), ...dest.stream.getAudioTracks()]);
    const rec = new MediaRecorder(stream, mime ? { mimeType: mime, videoBitsPerSecond: 3500000 } : undefined);
    const chunks = [];
    rec.ondataavailable = (e) => { if (e.data.size)
        chunks.push(e.data); };
    const done = new Promise((r) => { rec.onstop = () => r(); });
    rec.start(500);
    const begin = performance.now();
    await new Promise((resolve) => {
        const tick = () => {
            const el = (performance.now() - begin) / 1000;
            let i = starts.findIndex((s, n) => el >= s && (n === starts.length - 1 || el < starts[n + 1]));
            if (i < 0)
                i = el < starts[0] ? 0 : scenes.length - 1;
            drawFrame(ctx, { ...scenes[i], dur: durs[i] }, imgs[i], i, scenes.length, Math.max(0, el - starts[i]));
            onProgress(Math.min(1, el / total));
            if (el >= total)
                resolve();
            else
                setTimeout(tick, 33);
        };
        tick();
    });
    rec.stop();
    await done;
    await ac.close();
    const type = (rec.mimeType || mime || 'video/webm').split(';')[0];
    return { blob: new Blob(chunks, { type }), mime: type };
}
